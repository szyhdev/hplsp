cmake_minimum_required(VERSION 3.22)

project(05_fundamentals_of_network_programming
    LANGUAGES CXX
    VERSION 1.0
)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

set(common_source_file "${CMAKE_CURRENT_SOURCE_DIR}/common.cpp")
set(common_header_file "${CMAKE_CURRENT_SOURCE_DIR}/common.h")
include_directories(${CMAKE_CURRENT_SOURCE_DIR})

if(NOT DEFINED EXEFILE OR EXEFILE STREQUAL "all")
    file(GLOB SOURCE_FILES "*.cpp")
    list(REMOVE_ITEM SOURCE_FILES ${common_source_file})
    foreach(source_file ${SOURCE_FILES})
        get_filename_component(target_name ${source_file} NAME_WE)
        add_executable(${target_name} ${source_file} ${common_source_file} ${common_header_file})
    endforeach()
else()
    if(EXEFILE STREQUAL "01_byteorder")
        set(SOURCES 01_byteorder.cpp)
    elseif(EXEFILE STREQUAL "01_sockaddr")
        set(SOURCES 01_sockaddr.cpp)
    elseif(EXEFILE STREQUAL "02_nonreentrant")
        set(SOURCES 02_nonreentrant.cpp)
    elseif(EXEFILE STREQUAL "03_testlisten")
        set(SOURCES 03_testlisten.cpp)
    elseif(EXEFILE STREQUAL "05_testaccept")
        set(SOURCES 05_testaccept.cpp)
    elseif(EXEFILE STREQUAL "06_oobsend")
        set(SOURCES 06_oobsend.cpp)
    elseif(EXEFILE STREQUAL "07_oobrecv")
        set(SOURCES 07_oobrecv.cpp ${common_source_file} ${common_header_file})
    elseif(EXEFILE STREQUAL "07_oobrecv_atmark")
        set(SOURCES 07_oobrecv_atmark.cpp ${common_source_file} ${common_header_file})
    elseif(EXEFILE STREQUAL "09_reuse_address")
        set(SOURCES 09_reuse_address.cpp ${common_source_file} ${common_header_file})
    elseif(EXEFILE STREQUAL "10_set_send_buffer")
        set(SOURCES 10_set_send_buffer.cpp)
    elseif(EXEFILE STREQUAL "11_set_recv_buffer")
        set(SOURCES 11_set_recv_buffer.cpp ${common_source_file} ${common_header_file})
    elseif(EXEFILE STREQUAL "12_getdaytime")
        set(SOURCES 12_getdaytime.cpp ${common_source_file} ${common_header_file})
    else()
        message(FATAL_ERROR "Unknown EXEFILE: '${EXEFILE}'")
    endif()
    add_executable(${EXEFILE} ${SOURCES})
endif()
